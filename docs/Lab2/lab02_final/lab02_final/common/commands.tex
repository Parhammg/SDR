%----------------------------------------------------------------------
% Compatibility shim: bidi < 39 ships a multicol adaptation
% (multicol-xetex-bidi.def v1.2, 2019) that reads \mult@gfirstbox in the
% column-output routine. multicol >= 1.9f (TeX Live 2023) voids those
% scratch boxes during balancing and leaves the finished columns in
% \mult@firstbox, so the old shim prints only the last column and silently
% drops the rest. We restore the correct register. Guarded by bidi version
% so it stays inert on bidi >= 39 (the version the course targets).
\makeatletter
\@ifpackageloaded{bidi}{%
  \ifdim\bidi@version pt<39pt
    \def\LTR@column@boxes{%
        \process@cols\mult@firstbox{%
           \ifdim\dp\count@>\dimen\tw@
             \global\dimen\tw@\dp\count@ \fi
           \mc@col@status@write
           \box\count@
           \hss{\columnseprulecolor\vrule
                  \@width\columnseprule}\hss}%
         \ifdim\dp\mult@rightbox>\dimen\tw@
           \global\dimen\tw@\dp\mult@rightbox \fi
         \mc@lastcol@status@write
         \box\mult@rightbox}%
    \def\RTL@column@boxes{%
      \ifnum\col@number<\tw@
         \LTR@column@boxes
      \else
         \process@cols\mult@firstbox{%
           \hskip\hsize
           \hss{\columnseprulecolor\vrule
                  \@width\columnseprule}\hss}%
         \hskip\hsize
         \process@cols\mult@firstbox{%
           \ifdim\dp\count@>\dimen\tw@
             \global\dimen\tw@\dp\count@ \fi
           \hskip-\hsize
           \mc@col@status@write
           \box\count@
           \hskip-\hsize
           \hskip-\columnsep}%
         \ifdim\dp\mult@rightbox>\dimen\tw@
           \global\dimen\tw@\dp\mult@rightbox \fi
         \hskip-\hsize
         \mc@lastcol@status@write
         \box\mult@rightbox
         \hskip-\hsize
         \hskip\full@width
      \fi}%
  \fi
}{}%
\makeatother

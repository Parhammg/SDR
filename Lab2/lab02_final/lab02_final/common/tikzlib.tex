%----------------------------------------------------------------------
% 3) Helpers for text inside TikZ pictures
%    bidi typesets TikZ pictures left-to-right.  Therefore:
%      * Latin node text  -> type it directly
%      * Persian node text -> wrap it in \fa{...}
%----------------------------------------------------------------------
\newcommand{\fa}[1]{\rl{#1}}
% Without this, bidi inserts a right-to-left mark between "." or "-" and a
% following digit inside nodes, so that "2.048" would be shown as "048.2".
% Latin text in pictures is set in the Latin sans font with Latin digits.
\tikzset{every picture/.append style={execute at begin picture={\@nonlatinfalse\sffamily}}}
